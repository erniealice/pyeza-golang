{{/* ================================================================================ CALENDAR EVENT CHIP COMPONENT ================================================================================ Small event indicator used inside month-view day cells. Optionally also used as the canonical renderer for week/day-view event blocks when the caller passes positioning fields (.TopPct / .HeightPct). Renders a colored dot + truncated event name, status-coded. Optional tag dots (up to 3) appear to the right of the name; when a primary tag is present in week/day mode, a 4px left stripe in the tag's color is applied inline. Usage: {{template "calendar-event-chip" .}} Required data (CalendarEvent): .ID string — unique event ID .Name string — event display name .StartTime string — "10:00 AM" .EndTime string — "11:00 AM" (used by week/day variant's tooltip/aria) .Status string — "confirmed" | "tentative" | "cancelled" .DetailURL string — URL to navigate to on click Optional data: .HXTarget string — HTMX swap target (defaults to "#main-content" in month mode; "#calendar" in week/day mode) .Tags []{Name,Color} — tags assigned to the event, ordered by assignment position. Color is a hex string like "#E53E3E". When empty, chip renders exactly as before — no dots, no stripe, no behavioral change. Up to 3 dots are rendered; beyond that, 2 dots + "+N" badge. Tag names are folded into the chip's aria-label for screen readers. .TopPct float64 — week/day variant: absolute offset from column top (%) .HeightPct float64 — week/day variant: height as % of visible hours .IsCompact bool — week/day variant: single-line compact rendering ================================================================================ */}} {{define "calendar-event-chip"}} {{/* Detect week/day-view variant by presence of a non-zero height. */}} {{$isTimed := gt .HeightPct 0.0}} {{/* Build a tag-name suffix for aria-label when tags are present. */}} {{$tagNames := ""}} {{range $i, $tag := .Tags}} {{if eq $i 0}}{{$tagNames = $tag.Name}}{{else}}{{$tagNames = printf "%s, %s" $tagNames $tag.Name}}{{end}} {{end}} {{if $isTimed}} {{/* ── Week / Day variant ─────────────────────────────────────────── */}} {{$target := "#calendar"}} {{if .HXTarget}}{{$target = .HXTarget}}{{end}} {{$stripeColor := ""}} {{if gt (len .Tags) 0}}{{$stripeColor = (index .Tags 0).Color}}{{end}} {{.Name}} {{.StartTime}}{{if .EndTime}}–{{.EndTime}}{{end}} {{if gt (len .Tags) 0}} {{range $i, $tag := .Tags}} {{if lt $i 3}} {{end}} {{end}} {{if gt (len .Tags) 3}} +{{sub (len .Tags) 2}} {{end}} {{end}} {{else}} {{/* ── Month variant ──────────────────────────────────────────────── */}} {{$target := "#main-content"}} {{if .HXTarget}}{{$target = .HXTarget}}{{end}} {{.Name}} {{if gt (len .Tags) 0}} {{if gt (len .Tags) 3}} {{range $i, $tag := .Tags}} {{if lt $i 2}} {{end}} {{end}} +{{sub (len .Tags) 2}} {{else}} {{range .Tags}} {{end}} {{end}} {{end}} {{end}} {{end}}
